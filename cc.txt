2. parser.y
This file contains the logic for Task 2 (Tracking) and Task 3 (Two Registers). It maintains a state of what is currently in R0 and R1 to skip redundant LOAD instructions.

Code snippet
%{
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void yyerror(const char *s);
int yylex();

/* Task 2: Register Tracking State */
char reg0[20] = "";
char reg1[20] = "";

int temp_count = 0;
char* new_temp();

/* Task 3 logic: Generating code using two registers */
void gen_code(char* res, char* a1, char* op, char* a2) {
    printf("\n; --- TAC: %s = %s %s %s ---\n", res, a1, op, a2);

    // Load R0 (Arg1) only if not already there (Task 2)
    if (strcmp(reg0, a1) != 0) {
        printf("LOAD  R0, %s\n", a1);
        strcpy(reg0, a1);
    }

    // Load R1 (Arg2) for Multi-Register math (Task 3)
    if (strcmp(a2, "NULL") != 0) {
        if (strcmp(reg1, a2) != 0) {
            printf("LOAD  R1, %s\n", a2);
            strcpy(reg1, a2);
        }
    }

    // Task 1: Relational + Task 3: Reg-to-Reg Arithmetic
    if (strcmp(op, "+") == 0)      printf("ADD   R0, R1\n");
    else if (strcmp(op, "-") == 0) printf("SUB   R0, R1\n");
    else if (strcmp(op, "*") == 0) printf("MUL   R0, R1\n");
    else if (strcmp(op, "/") == 0) printf("DIV   R0, R1\n");
    else if (strcmp(op, "<") == 0 || strcmp(op, ">") == 0 || strcmp(op, "==") == 0) {
        printf("CMP   R0, R1\n");
        printf("JMP_%s TRUE_LABEL\n", op);
    }

    // Store Result
    printf("STORE %s, R0\n", res);
    strcpy(reg0, res); // R0 now contains the result
}
%}

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int yylex(void);
void yyerror(const char *s);


#define MAX_SYMBOLS 100

char *global_vars[MAX_SYMBOLS];
char *local_vars[MAX_SYMBOLS];
char *functions[MAX_SYMBOLS];

int global_count = 0;
int local_count = 0;
int function_count = 0;
int in_function = 0;


int exists(char *table[], int count, const char *name) {
    for (int i = 0; i < count; i++) {
      if (strcmp(table[i], name) == 0) return 1;
    }
    return 0;
}

void add_global(const char *name) {
    if (!exists(global_vars, global_count, name)) {
      global_vars[global_count++] = strdup(name);
    }
}

void add_local(const char *name) {
    if (!exists(local_vars, local_count, name)) {
        local_vars[local_count++] = strdup(name);
    }
}

void add_function(const char *name) {
    if (!exists(functions, function_count, name)) {
        functions[function_count++] = strdup(name);
    }
}
void clear_locals(void) {
    for (int i = 0; i < local_count; i++) {
        free(local_vars[i]);
    }
    local_count = 0;
}
void enter_function_scope(void) {
    in_function = 1;
    clear_locals();
}

void exit_function_scope(void) {
    in_function = 0;
    clear_locals();
}

int check_function_declared(const char *name) {
    if (exists(functions, function_count, name)) return 1;
    printf("Semantic Error: Function '%s' used before declaration\n", name);
    return 0;
}





int check_variable_visible(const char *name) {
    if (in_function) {
        if (exists(local_vars, local_count, name) ||
            exists(global_vars, global_count, name)) {
            return 1;
        }
    } else {
        if (exists(global_vars, global_count, name)) {
            return 1;
        }
    }

    printf("Semantic Error: '%s' undeclared\n", name);
    return 0;
}

#define MAX_ARRAYS 100
#define TYPE_INT   1
#define TYPE_FLOAT 2




typedef struct {
    char *name;
    int type;
    int size;
} ArraySymbol;




ArraySymbol arrays[MAX_ARRAYS];
int array_count = 0;

int find_array(const char *name) {
    for (int i = 0; i < array_count; i++) {
        if (strcmp(arrays[i].name, name) == 0) return i;
    }
    return -1;
}





const char *type_name(int t) {
    if (t == TYPE_INT) return "int";
    if (t == TYPE_FLOAT) return "float";
    return "unknown";
}




void add_array(const char *name, int type, int size) {
    if (find_array(name) == -1) {
        arrays[array_count].name = strdup(name);
        arrays[array_count].type = type;
        arrays[array_count].size = size;
        array_count++;
    }
}







int check_array_assignment(const char *name, int index, int value_type) {
    int pos = find_array(name);

    if (pos == -1) {
        printf("Semantic Error: '%s' undeclared\n", name);
        return 0;
    }

    if (index < 0) {
        printf("Semantic Error: Invalid array index (%d)\n", index);
        return 0;
    }

    if (index >= arrays[pos].size) {
        printf("Semantic Error: Array index %d out of bounds for '%s' of size %d\n",
               index, name, arrays[pos].size);
        return 0;
    }

    if (arrays[pos].type != value_type) {
        printf("Semantic Error: Type mismatch (%s vs %s)\n",
               type_name(arrays[pos].type), type_name(value_type));
        return 0;
    }

    return 1;
}

parser.y
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int yylex();
void yyerror(const char *s);

int tempCount = 1;


/* Create temp variable */
char* newTemp()
{
    char* temp = malloc(10);
    sprintf(temp, "t%d", tempCount++);
    return temp;
}


/* Check if string is numeric */
int isNumber(char* s)
{
    for(int i=0; s[i]; i++)
    {
        if(!isdigit(s[i]))
            return 0;
    }
    return 1;
}


/* Generate TAC */
void gen(char* result, char* a, char* op, char* b)
{
    printf("%s = %s %s %s\n", result, a, op, b);
}


/* Optimization + TAC generation */
char* optimize(char* left, char* op, char* right)
{
    /* ---------- Constant Folding ---------- */
    if(isNumber(left) && isNumber(right))
    {
        int x = atoi(left);
        int y = atoi(right);
        int ans;

        if(strcmp(op,"+")==0) ans = x+y;
        else if(strcmp(op,"-")==0) ans = x-y;
        else if(strcmp(op,"*")==0) ans = x*y;
        else if(strcmp(op,"/")==0) ans = x/y;

        char* result = malloc(20);
        sprintf(result,"%d",ans);

        printf("Optimized: %s %s %s -> %s\n",
                left, op, right, result);

        return result;
    }

    /* ---------- Algebraic Simplifications ---------- */

    if(strcmp(op,"+")==0)
    {
        if(strcmp(left,"0")==0) return strdup(right);
        if(strcmp(right,"0")==0) return strdup(left);
    }

    if(strcmp(op,"*")==0)
    {
        if(strcmp(left,"1")==0) return strdup(right);
        if(strcmp(right,"1")==0) return strdup(left);

        if(strcmp(left,"0")==0) return strdup("0");
        if(strcmp(right,"0")==0) return strdup("0");
    }

    /* ---------- Otherwise generate TAC ---------- */

    char* temp = newTemp();
    gen(temp,left,op,right);
    return temp;
}
